const COLLECTIONS=new Set(READABLE_COLLECTIONS),FIELD=/^[A-Za-z][A-Za-z0-9_]{0,63}$/,SELECTABLE=new Set(['state','nextAttemptAt']);

export interface ReadOnlyDoc{readonly id:string;data():Record<string,unknown>|undefined}
export interface ReadOnlyQuery{
  where(field:string,op:'==',value:string|number|boolean):ReadOnlyQuery;
  orderBy(field:'__name__'):ReadOnlyQuery;

export interface ReadOnlyOptions{journal?:(entry:{collection:string;op:'get'|'count';filters:number;docs:number})=>void;maxDocs?:number}

type RawQuery={where(...a:unknown[]):RawQuery;orderBy(...a:unknown[]):RawQuery;select(...a:unknown[]):RawQuery;limit(n:number):RawQuery;startAfter(c:unknown):RawQuery;get():Promise<{docs:Array<{id:string;data():Record<string,unknown>|undefined}>}>;count?:()=>{get():Promise<{data():{count:number}}>}};

  // `docs` is informational; only documents actually read (`reads`) count against the cap. A count() aggregation reads none.
  const account=(collection:string,op:'get'|'count',filters:number,docs:number,reads:number)=>{
    read+=reads;options.journal?.({collection,op,filters,docs});
    if(options.maxDocs!==undefined&&read>options.maxDocs)throw new Error('maximum documents exceeded');
  };
  const wrapDoc=(snap:{id:string;data():Record<string,unknown>|undefined}):ReadOnlyDoc=>{
    const doc=Object.freeze({id:snap.id,data:()=>snap.data()});raw.set(doc,snap);return doc;
  };
  const wrap=(collection:string,query:RawQuery,filters:number):ReadOnlyQuery=>Object.freeze({
    where(field:string,op:'==',value:string|number|boolean){

      if(typeof value!=='string'&&typeof value!=='number'&&typeof value!=='boolean')throw new TypeError('where: invalid value');

    },
    orderBy(...args:unknown[]){

    },
    limit(n:number){
      if(typeof n!=='number'||!Number.isSafeInteger(n)||n<1||n>1000)throw new RangeError('limit: invalid');
      return wrap(collection,query.limit(n),filters);
    },

      if(!snapshot)throw new TypeError('startAfter: not a cursor returned by this facade');

    },
  });

  });
